动态的语音播报
---

### 1、技术方案

- 实现代码如下

  ```js
  // 声音播放
  const testingSound = (params) => {
  
      if (!audioAllowed.value) {
          console.log('需要用户交互后才能播放音频')
          return
      }
  
      if ('speechSynthesis' in window) {
  
          if (window.speechSynthesis.speaking) {
              window.speechSynthesis.cancel()
          }
  
          const utterance = new SpeechSynthesisUtterance(params)
  
          utterance.lang = 'zh-CN'
          utterance.rate = 1.5
          utterance.pitch = 1
  
          utterance.onstart = () => console.log('语音开始播放')
          utterance.onend = () => console.log('语音播放结束')
          utterance.onerror = (e) => console.error('语音播放错误:', e)
  
          window.speechSynthesis.speak(utterance)
  
      } else {
  
          console.error('您的浏览器不支持Web Speech API语音')
      }
  }
  ```
